            mainModel.finder.formOpen(false);
            $(el, th).show();

                mainModel.finder.formOpen(true);
            }

        hideAllMenu('.editRoute', $(this).parent());

        $(this).addClass('showInlineRoutes');

        $("div[class*='markk']").each(function() {
           // if ($(this).attr('id') === m)
            if($(this).hasClass(m))

            else
                $(this).addClass("marker_off");
        });
    });

    route_bl_selector.live('mouseout', function() {
        $(this).removeClass('showInlineRoutes');
        var m = "markk" + $(this).attr('tt') + "_" + $(this).attr('marsh');
        $("div[class*='markk']").each(function() {
